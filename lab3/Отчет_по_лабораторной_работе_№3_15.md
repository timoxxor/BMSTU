## ЧАСТЬ 1: Классы и перегрузка операций в Си++

### 1. Цель работы
Овладение практическими навыками разработки программ на C++, использующих классы, динамическое выделение памяти, конструкторы/деструкторы, семантику перемещения (Move Semantics), а также перегрузку стандартных операций.

### 2. Условие задачи (Вариант 15)
1. Разработать класс `Vector`, задающий динамический вектор элементов `int` размерности $n$.
2. **Поля класса:** указатель на динамический массив (`int* data`), размер вектора (`int size`).
3. **Методы и конструкторы:**
   * Конструктор без параметров (размер 0, указатель `nullptr`);
   * Конструктор на основе одномерного массива размерности $n$;
   * Конструктор копирования (глубокое копирование памяти);
   * Конструктор перемещения (перенос указателя);
   * Деструктор.
4. **Перегрузка операций:**
   * Индексирование `[]`;
   * Присваивание с копированием `operator=`;
   * Присваивание с перемещением `operator=`;
   * Потоковый ввод `>>` и вывод `<<`.
5. **Индивидуальная операция (Вариант 15):**
   * Побитовая операция `^` (XOR) для двух векторов одинаковой размерности:
     $$\text{result}[i] = \text{vec}_1[i] \oplus \text{vec}_2[i]$$
   * При несовпадении размерностей — выброс исключения `std::invalid_argument`.
6. Продемонстрировать разницу работы конструкторов/присваиваний с копированием и с перемещением. Входные данные читать из `input.txt`, результаты выводить в `output.txt`.

#### `src/Vector.cpp`:
```cpp
#include "Vector.hpp"
#include <utility>

Vector::Vector() : data(nullptr), size(0) {}

Vector::Vector(int n, int initVal) : data(nullptr), size(0) {
    if (n < 0) {
        throw std::invalid_argument("Vector size cannot be negative");
    }
    if (n > 0) {
        size = n;
        data = new int[size];
        for (int i = 0; i < size; ++i) {
            data[i] = initVal;
        }
    }
}

Vector::Vector(const int* arr, int n) : data(nullptr), size(0) {
    if (n < 0) {
        throw std::invalid_argument("Array size cannot be negative");
    }
    if (n > 0 && arr != nullptr) {
        size = n;
        data = new int[size];
        for (int i = 0; i < size; ++i) {
            data[i] = arr[i];
        }
    }
}

Vector::Vector(const Vector& other) : data(nullptr), size(other.size) {
    if (size > 0 && other.data != nullptr) {
        data = new int[size];
        for (int i = 0; i < size; ++i) {
            data[i] = other.data[i];
        }
    }
}

Vector::Vector(Vector&& other) noexcept : data(other.data), size(other.size) {
    other.data = nullptr;
    other.size = 0;
}

Vector::~Vector() {
    delete[] data;
    data = nullptr;
    size = 0;
}

Vector& Vector::operator=(const Vector& other) {
    if (this == &other) {
        return *this;
    }

    int* newData = nullptr;
    if (other.size > 0 && other.data != nullptr) {
        newData = new int[other.size];
        for (int i = 0; i < other.size; ++i) {
            newData[i] = other.data[i];
        }
    }

    delete[] data;
    data = newData;
    size = other.size;

    return *this;
}

Vector& Vector::operator=(Vector&& other) noexcept {
    if (this == &other) {
        return *this;
    }

    delete[] data;

    data = other.data;
    size = other.size;

    other.data = nullptr;
    other.size = 0;

    return *this;
}

int& Vector::operator[](int index) {
    if (index < 0 || index >= size) {
        throw std::out_of_range("Vector index out of range");
    }
    return data[index];
}

const int& Vector::operator[](int index) const {
    if (index < 0 || index >= size) {
        throw std::out_of_range("Vector index out of range");
    }
    return data[index];
}

Vector Vector::operator^(const Vector& other) const {
    if (size != other.size) {
        throw std::invalid_argument("Vectors must have the same dimension for XOR operation (^)");
    }

    Vector result(size);
    for (int i = 0; i < size; ++i) {
        result.data[i] = data[i] ^ other.data[i];
    }
    return result;
}

int Vector::getSize() const {
    return size;
}

const int* Vector::getData() const {
    return data;
}

std::ostream& operator<<(std::ostream& os, const Vector& vec) {
    os << "[";
    for (int i = 0; i < vec.size; ++i) {
        os << vec.data[i];
        if (i + 1 < vec.size) {
            os << ", ";
        }
    }
    os << "] (size: " << vec.size << ")";
    return os;
}

std::istream& operator>>(std::istream& is, Vector& vec) {
    int newSize = 0;
    if (is >> newSize) {
        if (newSize < 0) {
            is.setstate(std::ios::failbit);
            return is;
        }

        int* newElements = nullptr;
        if (newSize > 0) {
            newElements = new int[newSize];
            for (int i = 0; i < newSize; ++i) {
                if (!(is >> newElements[i])) {
                    delete[] newElements;
                    is.setstate(std::ios::failbit);
                    return is;
                }
            }
        }

        delete[] vec.data;
        vec.data = newElements;
        vec.size = newSize;
    }
    return is;
}
```

```cpp
#include "Vector2D.hpp"
#include <cmath>
#include <iomanip>

Vector2D::Vector2D(double x, double y) : x(x), y(y) {}

double Vector2D::getX() const {
    return x;
}

double Vector2D::getY() const {
    return y;
}

void Vector2D::setX(double newX) {
    x = newX;
}

void Vector2D::setY(double newY) {
    y = newY;
}

double Vector2D::length() const {
    return std::sqrt(x * x + y * y);
}

void Vector2D::print(std::ostream& os) const {

}
```

#### `src/Vector3D.cpp`:
```cpp
#include "Vector3D.hpp"
#include <cmath>
#include <iomanip>

Vector3D::Vector3D(double x, double y, double z) : Vector2D(x, y), z(z) {}

double Vector3D::getZ() const {
    return z;
}

void Vector3D::setZ(double newZ) {
    z = newZ;
}

double Vector3D::length() const {
    double len2D = Vector2D::length();
    return std::sqrt(len2D * len2D + z * z);
}

void Vector3D::print(std::ostream& os) const {
    os << "Vector3D(" << std::fixed << std::setprecision(2) << x << ", " << y << ", " << z << ")"
       << " len: " << std::setprecision(4) << length()
       << " [2D proj: " << Vector2D::length() << "]";
}
```

#### `src/main.cpp`:
```cpp
#include <iostream>
#include <vector>
#include <iomanip>
#include "Vector2D.hpp"
#include "Vector3D.hpp"

int main() {
    Vector2D v2d(3.0, 4.0);
    Vector3D v3d(1.0, 2.0, 2.0);

    std::cout << "v2d: ";
    v2d.print();
    std::cout << "\n";

    std::cout << "v3d: ";
    v3d.print();
    std::cout << "\n\n";

    std::cout << "Static polymorphism:\n";
    std::cout << "  v3d len: " << v3d.length() << "\n";
    std::cout << "  v3d base len: " << v3d.Vector2D::length() << "\n\n";

    std::cout << "Dynamic polymorphism:\n";
    Vector2D* p1 = &v2d;
    Vector2D* p2 = &v3d;
    std::cout << "  p1 (v2d): ";
    p1->print();
    std::cout << "\n";
    std::cout << "  p2 (v3d): ";
    p2->print();
    std::cout << " (len: " << p2->length() << ")\n\n";

    Vector2D obj1(6.0, 8.0);
    Vector3D obj2(2.0, 3.0, 6.0);
    Vector2D obj3(0.0, 12.0);
    Vector3D obj4(4.0, 4.0, 7.0);

    std::vector<Vector2D*> vecList = { &v2d, &v3d, &obj1, &obj2, &obj3, &obj4 };

    std::cout << "Vector list:\n";
    double totalLength = 0.0;
    for (size_t i = 0; i < vecList.size(); ++i) {
        std::cout << "  [" << i << "] ";
        vecList[i]->print();
        std::cout << "\n";
        totalLength += vecList[i]->length();
    }

    std::cout << "\nTotal length: " << std::fixed << std::setprecision(4) << totalLength << "\n";
    return 0;
}
```
